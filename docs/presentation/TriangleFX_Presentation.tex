\begin{frame}{1. Problem Statement}
  \begin{columns}[T]
    \column{0.52\textwidth}
      \begin{block}{The problem}
        Given \textbf{three lines} in the plane, decide whether they bound a
        real triangle. If they do, compute and \textbf{draw} it.
      \end{block}
      \vspace{2pt}
      Requirements
      \begin{itemize}
        \item Accept input as the matrix system $A\mathbf{x}=\mathbf{b}$
        \item Find the three vertices (pairwise intersections)
        \item Reject impossible geometry with a \emph{clear reason}
        \item Show area, perimeter, sides and centroid
        \item Render on a scaled coordinate plane
      \end{itemize}
    \column{0.44\textwidth}
      \begin{alertblock}{Inputs that must fail gracefully}
        \begin{itemize}
          \item Two lines parallel
          \item Two lines identical
          \item All three lines through one point
          \item Vertices collinear (zero area)
          \item Malformed / missing numbers
        \end{itemize}
      \end{alertblock}
      \begin{exampleblock}{Example}
        $x+y=8,\; x-y=2,\; x=1$\\[2pt]
        $\Rightarrow$ vertices $(5,3),(1,-1),(1,7)$, area $16$
      \end{exampleblock}
  \end{columns}
\end{frame}

    \node[pstep,right=of b] (c) {\textbf{Generalise}\\ rows of $A\mathbf{x}=\mathbf{b}$};
    \node[pstep,below=of a] (d) {\textbf{Solve}\\ $2\times2$ per line pair};
    \node[pstep,right=of d] (e) {\textbf{Verify}\\ distinct, non-flat};
    \node[pstep,right=of e] (f) {\textbf{Present}\\ scale and draw};
    \draw[arr] (a)--(b); \draw[arr] (b)--(c);
    \draw[arr] (c.south) -- ++(0,-2.5mm) -| ($(d.north)+(0,0)$);
    \draw[arr] (d)--(e); \draw[arr] (e)--(f);
  \end{tikzpicture}
  \end{center}
  \vspace{-2mm}
  \begin{columns}[T]
    \column{0.48\textwidth}
      \textbf{Key decisions}
      \begin{itemize}
        \item Matrix input is unambiguous; no free-text algebra needed in the UI
        \item Keep math \textbf{independent of JavaFX} so it is unit-testable
        \item Treat invalid geometry as a first-class, explained outcome
      \end{itemize}
    \column{0.48\textwidth}
      \textbf{Alternatives considered}
      \begin{itemize}
        \item Parsing typed equations (\texttt{x + y = 8}): kept as an optional
              \texttt{EquationParser}, not the primary path
        \item Solving a $3\times2$ system directly: over-determined in general,
              so we solve \emph{pairs} instead
      \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{3. The Mathematics}
  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{block}{1. Matrix Formulation ($A\mathbf{x} = \mathbf{b}$)}
        \[
          \begin{bmatrix} a_{11} & a_{12} \\ a_{21} & a_{22} \\ a_{31} & a_{32} \end{bmatrix}
          \begin{bmatrix} x \\ y \end{bmatrix} =
          \begin{bmatrix} b_1 \\ b_2 \\ b_3 \end{bmatrix}
          \quad \begin{subarray}{l} \text{\scriptsize Line 1} \\ \text{\scriptsize Line 2} \\ \text{\scriptsize Line 3} \end{subarray}
        \]
      \end{block}
      \vspace{2pt}
      \begin{block}{2. Vertex Intersection (Cramer's Rule)}
        \footnotesize
        For lines $L_1, L_2$ ($D = a_{11}a_{22} - a_{21}a_{12} \neq 0$):
        \[
          x = \frac{b_1 a_{22} - b_2 a_{12}}{D}, \quad
          y = \frac{a_{11} b_2 - a_{21} b_1}{D}
        \]
        Gives corners $P_{12}, P_{23}, P_{31}$ ($D=0 \implies \text{parallel}$).
      \end{block}

    \column{0.48\textwidth}
      \begin{block}{3. Triangle Properties}
        \footnotesize
        \begin{itemize}\setlength{\itemsep}{3pt}\setlength{\parskip}{0pt}
          \item \textbf{Area (Shoelace):}\\
                $\Delta = \tfrac{1}{2}\bigl|x_1(y_2-y_3) + x_2(y_3-y_1) + x_3(y_1-y_2)\bigr|$
          \item \textbf{Centroid (Balance):}\\
                $G = \bigl(\tfrac{x_1 + x_2 + x_3}{3}, \, \tfrac{y_1 + y_2 + y_3}{3}\bigr)$
          \item \textbf{Perimeter:}\\
                $p = a + b + c \quad (d = \sqrt{\Delta x^2 + \Delta y^2})$
        \end{itemize}
      \end{block}
      \vspace{2pt}
      \begin{exampleblock}{4. Geometric Validation}
        \footnotesize
        \textbf{Valid} $\iff D \neq 0$, distinct corners, and $\Delta \ge 10^{-6}$.\\[1pt]
        \scriptsize (Rejects parallel, coincident, concurrent, or flat lines)
      \end{exampleblock}
  \end{columns}
\end{frame}

  \vspace{2mm}
  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{itemize}
        \item \textbf{Inward Dependencies}: \texttt{model} is pure domain logic, independent of UI or parsing
        \item \textbf{Separation of Concerns}: \texttt{ui} handles view/events; \texttt{parser} validates inputs
      \end{itemize}
    \column{0.48\textwidth}
      \begin{itemize}
        \item \textbf{Headless Testing}: \texttt{geometry}/\texttt{model} are 100\% unit-tested without a display
        \item \textbf{Robust Exceptions}: \texttt{GeometryException} surfaces clear failure reasons to the UI
      \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{5. Solution Walk-through: Validation Pipeline}
  \begin{columns}[T]
    \column{0.40\textwidth}
      \small
      \begin{enumerate}
        \item \textbf{Pair check} (3 pairs): identical or parallel $\to$ error
        \item \textbf{Intersect} each pair (Cramer)
        \item \textbf{Concurrency}: two vertices within tolerance $\to$ error
        \item \textbf{Area}: if $\Delta<10^{-6}\to$ error
        \item Return the verified \texttt{Triangle}
      \end{enumerate}
      \vspace{2pt}
      Every failure carries a human-readable message naming the
      offending lines.
    \column{0.57\textwidth}
\begin{lstlisting}[basicstyle=\ttfamily\fontsize{6.5}{8}\selectfont,breaklines=false,xleftmargin=2pt,xrightmargin=2pt]
Triangle buildTriangle(Line l1, Line l2, Line l3)
    throws GeometryException {
  // 1. reject identical / parallel pairs
  checkPair(l1, l2, 1, 2);
  checkPair(l2, l3, 2, 3);
  checkPair(l3, l1, 3, 1);

  // 2. the three vertices
  Point p12 = l1.intersectionWith(l2).orElseThrow(..);
  Point p23 = l2.intersectionWith(l3).orElseThrow(..);
  Point p31 = l3.intersectionWith(l1).orElseThrow(..);

  // 3. concurrency: coincident vertices
  if (p12.equalsWithTolerance(p23, TOL) || ..)
    throw new GeometryException("concurrent");

  // 4. degenerate (near-zero area)
  Triangle t = new Triangle(p12, p23, p31, l1, l2, l3);
  if (t.getArea() < MIN_AREA)
    throw new GeometryException("collapsed");
  return t;
}
\end{lstlisting}
  \end{columns}
\end{frame}

\begin{frame}{6. Rendering: World-to-Screen Mapping}
  \begin{columns}[T]
    \column{0.52\textwidth}
      The canvas fits the triangle to any window size using a single
      uniform scale, so shapes are never distorted:
      \[
        X_{\text{px}} = \tfrac{w}{2} + (x - x_c)\,s
      \]
      \[
        Y_{\text{px}} = \tfrac{h}{2} - (y - y_c)\,s
      \]
      \begin{itemize}
        \item $(x_c,y_c)$: centre of the bounding box; $s$: pixels per unit
        \item The minus sign flips $y$ (screen $y$ grows downward)
        \item Grid, axes, lines, fill, and labelled vertices drawn in order
      \end{itemize}
    \column{0.44\textwidth}
      \begin{center}
      \begin{tikzpicture}[scale=0.42,>=Stealth]
        \draw[step=1,gray!40,very thin] (-1,-2) grid (7,8);

\begin{frame}{7. Design Patterns and Principles in the Codebase}
  \footnotesize
  \renewcommand{\arraystretch}{1.12}
  \begin{tabularx}{\textwidth}{@{}p{3.1cm}p{3.6cm}X@{}}
    \toprule
    \textbf{Pattern / principle} & \textbf{Where} & \textbf{Why it helps}\\
    \midrule
    Facade / service layer & \texttt{GeometryService} & One entry point hides parsing, pairing, intersection and validation\\
    Immutable value objects & \texttt{Point}, \texttt{Line}, \texttt{Triangle} & \texttt{final} fields: thread-safe, safe to share with the UI\\
    \texttt{Optional} for ``no result'' & \texttt{Line.intersectionWith} & Callers must handle parallel lines; no \texttt{null}\\
    Fail-fast + exception translation & \texttt{Line}, \texttt{GeometryService} & \texttt{Line} rejects $A=B=0$; service rewraps as a domain error\\
    Static utility classes & \texttt{MatrixParser}, \texttt{EquationParser} & Private constructors; stateless, easy to test\\
    Observer (JavaFX listeners) & \texttt{TriangleCanvas} & Size listeners call \texttt{redraw()} on resize\\
    Separation of concerns & \texttt{model}, \texttt{geometry}, \texttt{ui} & Logic testable without a display\\
    \bottomrule
  \end{tabularx}
\end{frame}

\begin{frame}[fragile]{8. Engineering Quality}
  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{block}{Numerical robustness}
        \begin{itemize}
          \item Tolerances ($10^{-9}$ for parallelism, $10^{-6}$ for area/points) instead of exact \texttt{==}
          \item Normalises $-0.0$ in intersection results
          \item \texttt{Math.hypot} for the ``$A,B$ not both zero'' check
        \end{itemize}
      \end{block}
      \begin{block}{Two error families}
        \texttt{ParseException} (bad text) vs.\ \texttt{GeometryException}
        (bad geometry): the UI reports each differently.
      \end{block}
    \column{0.48\textwidth}
      \begin{exampleblock}{Tests (JUnit 5)}
        \begin{itemize}
          \item \texttt{GeometryServiceTest}
          \item \texttt{EquationParserTest}
          \item \texttt{MatrixParserTest}
        \end{itemize}
        Cover valid triangles, parallel/coincident/concurrent/degenerate
        inputs and parser edge cases.
      \end{exampleblock}
      \begin{block}{Documentation and build}
        Javadoc on the public API, Maven build, dark-theme UI;
        design docs in \texttt{docs/}.
      \end{block}
  \end{columns}
\end{frame}
